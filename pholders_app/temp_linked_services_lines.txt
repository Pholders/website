196	  Widget _buildSection(String title, List<_ServiceItem> items) {
197	    return Column(
198	      crossAxisAlignment: CrossAxisAlignment.start,
199	      children: [
200	        Text(
201	          title,

203	            fontSize: 15,

215	          child: Column(

217	              items.length,

223	                      const Divider(

240	  Widget _buildServiceRow(_ServiceItem item) {
241	    final isConnected = _connected[item.id] ?? false;
242	
243	    return Padding(
244	      padding: const EdgeInsets.symmetric(horizontal: 16, vertical: 14),
245	      child: Row(
246	        children: [
247	          Container(
248	            width: 42,
249	            height: 42,
250	            decoration: BoxDecoration(
251	              color: const Color(0x14309CA3),
252	              borderRadius: BorderRadius.circular(12),
253	            ),
254	            child: Image.asset(
255	              item.iconAsset,
256	              width: 20,
257	              height: 20,
258	              fit: BoxFit.contain,
259	            ),
260	          ),
261	          const SizedBox(width: 14),
262	          Expanded(
263	            child: Column(
264	              crossAxisAlignment: CrossAxisAlignment.start,
265	              children: [
266	                Text(
267	                  item.name,
268	                  style: const TextStyle(
269	                    fontSize: 15,
270	                    fontWeight: FontWeight.w600,
271	                    color: _textDark,
272	                  ),
273	                ),
274	                if (item.subtitle != null) ...[
275	                  const SizedBox(height: 4),
276	                  Text(
277	                    item.subtitle!,
278	                    style: const TextStyle(
279	                      fontSize: 12,
280	                      color: _textMuted,
281	                    ),
282	                  ),
283	                ],
284	              ],
285	            ),
286	          ),
287	          const SizedBox(width: 14),
288	          InkWell(
289	            onTap: () => setState(() => _connected[item.id] = !isConnected),
290	            borderRadius: BorderRadius.circular(12),
291	            child: Container(
292	              padding: const EdgeInsets.symmetric(horizontal: 22, vertical: 12),
293	              decoration: BoxDecoration(
294	                color: isConnected ? const Color(0x15309CA3) : Colors.white,
295	                borderRadius: BorderRadius.circular(12),
296	                border: Border.all(
297	                  color: isConnected ? _tealBorder : const Color(0xFFE5E7EB),
298	                ),
299	              ),
300	              child: Text(
301	                isConnected ? 'Disconnect' : 'Connect',
302	                style: TextStyle(
303	                  fontSize: 13,
304	                  fontWeight: FontWeight.w700,
305	                  color: isConnected ? _teal : _textDark,
306	                ),
307	              ),
308	            ),
309	          ),
310	        ],
311	      ),
312	    );
313	  }
